# Compare the two listings. readdir() order is not guaranteed, so sort both,
# and write the results outside the folder so the result files do not list themselves.
./dir_list_lib | sort > /tmp/lib.txt
./dir_list_sys | sort > /tmp/sys.txt
diff /tmp/lib.txt /tmp/sys.txt && echo "OUTPUTS IDENTICAL"
